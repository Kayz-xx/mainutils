client.modlogs = async function({ Member, Action, Color, Reason }, message) {
    const data = await modLogsSchema.findOne({ Guild: message.guild.id });
    if(!data) return;

    const channel = message.guild.channel.cache.get(data.Channel);
    const logsEmbed = new messageEmbed()
    .setColor(Color)
    .setDescription(`Reason: ${Reason || 'No reason!'}`)
    .addField('Member', `${Member.user.tag} (${Member.id})`)
    .setTitle(`Donation added: ${Action}`)

    channel.send(logsEmbed)
}

module.exports = {
    commands: ['rolecheck'],
    maxArgs: 1,
callback: async (message, arguments) => {
    const check = await db.has(`autorole-${message.guild.id}`);
    if(check === false) return message.reply('No donation roles have been set!');
    const role = await db.length(`autorole-${message.guild.id}`)
    message.reply(`The Donation role is<@${role}>`)
}

const { Client, Message, MessageEmbed } = require('discord.js')
const db = require('../../mongo')

module.exports = {
    commands: ['autorole'],
    maxArgs: 1,
callback: async (message, arguments, args) => {
        if(!message.member.hasPermission("MANAGE_ROLES")) return message.channel.send('You need manage roles permissions to use this command')
        const role = message.mentions.roles.first() || message.guild.roles.cache.get(args[0])
        if(!role) return message.channel.send('Role is not valid')

        await db.set(`autorole-${message.guild.id}`, role.id);
        message.reply(`${role.name} is set as donation role`)
    }   
}
