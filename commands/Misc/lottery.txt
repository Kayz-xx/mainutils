const { MessageEmbed } = require('discord.js');
const { db } = require('../../firebase');
let playerList = [];
module.exports = {
	name: 'lotterystart',
	aliases: ['enter', 'lotteryend'],
	cooldown: '0',
	permissions: [],
	description:
		'Lottery System!, (lotterystart- starts the lottery) (enter- allows the host to add users) (lotteryend- ends the lottery and determines a winner!)',
	category: 'Misc',

	async execute(client, message, cmd, args) {
		if (cmd === 'lotterystart') {
			let data =
				(await db
					.ref(`Lottery System/${message.guild.id}/Lottery`)
					.once('value')
					.then((snapshot) => snapshot.val())) || [];
			if (data.status === 'true') {
				return message.channel.send({embeds: 
					[new MessageEmbed()
						.setTitle('An Error Occured <:sim:860034795169251358>')
						.setDescription('A lottery is already being hosted!')
						.setFooter(`Run \`lotteryend\` to end the lottery`)
						.setTimestamp()
						.setColor('CE1212')]
				});
			} else {
				const prize = args.slice(0).join(' ');
				if (!prize) {
					return message.channel.send({content: 'Please specify a prize!'});
				}
				message.channel.send({embeds:
					[new MessageEmbed()
						.setAuthor(
							'New Lottery!',
							message.author.avatarURL({ dynamic: true })
						)
						.setTitle(
							`${message.author.tag} started a lottery for ${prize}! <a:im4:858370157890371595>`
						)
						.setThumbnail(
							'https://cdn.discordapp.com/icons/764885367160700958/a_38503e9dec18ac442fecaad24a3d07c0.gif?size=1024'
						)
						.setTimestamp()
						.setColor('5C33F6')
						]});
				lotteryBool = 'true';
				let host = message.member.id;
				db.ref(`Lottery System/${message.guild.id}/Lottery/status`).set(
					lotteryBool
				);
				db.ref(`Lottery System/${message.guild.id}/Lottery/prize`).set(
					prize
				);
				db.ref(`Lottery System/${message.guild.id}/Lottery/host`).set(
					host
				);
			}
		}
		if (cmd === 'enter') {
			let data =
				(await db
					.ref(`Lottery System/${message.guild.id}/Lottery`)
					.once('value')
					.then((snapshot) => snapshot.val())) || [];
			const members = args.slice(0);
			if (!members)
				return message.channel.send({content:
					`enter [members]\n      ^^^^^^^^\nMissing Argument(Atleast 1 member required)`
				});
			if (data.status === 'false') {
				message.channel.send({embeds: [
					new MessageEmbed()
						.setTitle('An Error Occured <:sim:860034795169251358>')
						.setDescription(
							'There is no lottery being hosted at the moment.'
						)
						.setFooter(`Ask staff to host a lottery.`)
						.setTimestamp()
						.setColor('CE1212')
						]});
			} else if (message.member.id !== data.host)
				return message.channel.send({embeds: [
					new MessageEmbed()
						.setTitle('An Error Occured <:sim:860034795169251358>')
						.setDescription('Only host can enter participants')
						.setFooter(`Ask host to enter.`)
						.setTimestamp()
						.setColor('CE1212')
				]});
			else if (data.status === 'true') {
				if (playerList.includes(members)) {
					return message.channel.send({embeds: [
						new MessageEmbed()
							.setTitle(
								'An Error Occured <:sim:860034795169251358>'
							)
							.setDescription(
								'You have already entered the lottery'
							)
							.setFooter(`Only 1 entry is allowed per lottery`)
							.setTimestamp()
							.setColor('CE1212')
							]});
				} else {
					playerList = playerList.concat(members);
					message.channel.send({embeds: [
						new MessageEmbed()
							.setAuthor(
								'Lottery',
								message.author.avatarURL({ dynamic: true })
							)
							.setDescription(
								`Host has entered ${members} the lottery for **${data.prize}**!`
							)
							.setFooter('Good Luck')
							.setTimestamp()
							.setColor('5C33F6')
					]});

					db.ref(
						`Lottery System/${message.guild.id}/Lottery/Entries`
					).set(playerList);
					return;
				}
			}
		}

		if (cmd === 'lotteryend') {
			let data =
				(await db
					.ref(`Lottery System/${message.guild.id}/Lottery`)
					.once('value')
					.then((snapshot) => snapshot.val())) || [];
			if (data.status === 'false') {
				return message.channel.send({embeds: [
					new MessageEmbed()
						.setTitle('An Error Occured <:sim:860034795169251358>')
						.setDescription('No lottery is being held!')
						.setFooter(`Use \`enter\` to join the lottery`)
						.setTimestamp()
						.setColor('CE1212')
				]});
			} else if (!data.Entries) {
				return (
					message.channel.send({embeds: [
						new MessageEmbed()
							.setTitle(
								'An Error Occured <:sim:860034795169251358>'
							)
							.setDescription('No one entered the lottery!')
							.setFooter(`Use \`enter\` to join the lottery`)
							.setTimestamp()
							.setColor('CE1212')
					]}),
					(lotteryBool = 'false'),
					db
						.ref(
							`Lottery System/${message.guild.id}/Lottery/status`
						)
						.set(lotteryBool)
				);
			} else if (data.Entries.length >= 1) {
				lotteryBool = 'false';
				db.ref(`Lottery System/${message.guild.id}/Lottery/status`).set(
					lotteryBool
				);

				const realWinner =
					data.Entries[
						Math.floor(Math.random() * data.Entries.length)
					];
				delete data.Entries.splice(0, data.Entries.length);
				db.ref(
					`Lottery System/${message.guild.id}/Lottery/Entries`
				).set(data.Entries);

				const user = await client.users.fetch(realWinner);
				user.send({embeds: [
					new MessageEmbed()
						.setDescription(
							`Congratulations you have won the lottery for :\n **${data.prize}**`
						)
						.addField(`\u200B`, `[Lottery](${message.url})`)
						.setFooter('Your prize will be payed out soon.')
						.setTimestamp()
						.setColor('5C33F6')
				]});
				message.channel.send({embeds: [
					new MessageEmbed()
						.setTitle('Congratulations! <a:im6:860038641841602581>')
						.setDescription(`<@${realWinner}> has won the lottery!`)
						.setFooter('Your prize will be payed out soon.')
						.setTimestamp()
						.setColor('5C33F6')
				]});
			} else if (message.member.id !== data.host) {
				return message.channel.send({embeds: [
					new MessageEmbed()
						.setTitle('An Error Occured <:sim:860034795169251358>')
						.setDescription('Only host can end lottery')
						.setFooter(`Ask host to enter.`)
						.setTimestamp()
						.setColor('CE1212')
				]});
			}
			playerList = [];
		}
	},
};
